% Manually numbered theorem-like environments, so numbers can match the
% lecturer's master documents. Usage:
%   \begin{definition}{2.3}            ...  \end{definition}  -> Definition 2.3.
%   \begin{theorem}{4.1}[Compactness]  ...  \end{theorem}     -> Theorem 4.1 (Compactness).
% \label works as usual, so \ref{...} prints the number you typed.
\newcommand{\manualthmname}{}
\newcommand{\manualthmnum}{}
\newtheorem{manualthm}{\manualthmname}
\renewcommand{\themanualthm}{\manualthmnum}
\newcommand{\newmanualtheorem}[2]{%
  \NewDocumentEnvironment{#1}{m O{}}
    {\renewcommand{\manualthmname}{#2}%
     \renewcommand{\manualthmnum}{##1}%
     \if\relax\detokenize{##2}\relax
       \begin{manualthm}% no title (or empty [])
     \else
       \begin{manualthm}[##2]%
     \fi}
    {\end{manualthm}}%
}
\newmanualtheorem{definition}{Definition}
\newmanualtheorem{example}{Example}
\newmanualtheorem{theorem}{Theorem}
\newmanualtheorem{lemma}{Lemma}
\newmanualtheorem{prop}{Proposition}
\newmanualtheorem{corollary}{Corollary}
 
% Manually numbered headings, so numbers can match the lecturer's master documents. Usage:
%   \msection{1}{Combinator Calculus}
%   \msubsection{1.1}{Combinator Algebra}
% \label works right after the heading, and \ref prints the number you typed.
\newcommand{\msection}[2]{%
  \renewcommand{\thesection}{#1}%
  \refstepcounter{section}%
  \section*{#1\quad #2}%
  \addcontentsline{toc}{section}{\protect\numberline{#1}#2}}
\newcommand{\msubsection}[2]{%
  \renewcommand{\thesubsection}{#1}%
  \refstepcounter{subsection}%
  \subsection*{#1\quad #2}%
  \addcontentsline{toc}{subsection}{\protect\numberline{#1}#2}}
