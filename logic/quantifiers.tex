\subsection{Predicates \& Quantifiers}

\begin{definition}[first-order language]
    The symbols are of two kinds:
    \begin{enumerate}
        \item The logical symbols, the same in every language: $\to, \land, \lor, \bot$; the quantifiers $\forall,
        \exists$; variables $x, y, z, \dots$
        \item The signature, which changes from language to language: constant symbols $c, d, \dots$; function symbols
        $f, g, \dots$; predicate symbols $P, Q, \dots$, each with an arity.
    \end{enumerate}
\end{definition}

\begin{definition}[terms] \leavevmode
    \begin{enumerate}
        \item Every variable is a term;
        \item Every constant symbol is a term;
        \item If $f$ has arity $n$ and $t_1, \dots, t_n$ are terms, $f(t_1, \dots, t_n)$ is a term;
        \item Nothing else is a term.
    \end{enumerate}
\end{definition}
