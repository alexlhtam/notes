\section{Combinator Calculus}

\subsection{Combinator Algebra}

\begin{definition}{1.1}[terms, and combinators]
  Fix two \textit{constants}, $\cK$ and $\cS$, and a stock of \textit{variables} $x, y, z, f, g, \dots$. The
  \textit{terms} are defined inductively:
  \begin{enumerate}
    \item Each constant, $\cK$ and $\cS$, is a term.
    \item Each variable $x$ is a term.
    \item If $M$ and $N$ are terms, then $(M \, N)$ is a term: the \textit{application} of $M$ to $N$.
    \item Nothing else is a term.
  \end{enumerate}
  A term without any variables is a \textit{combinator}.

  In one line, with $M$ and $N$ standing for terms already built, \[M,N ::= \cK \, | \, \cS \, | \, x \, | \, (M \, N)\]

  Writing out every bracket gets tiresome, so we adopt the convention where \textbf{application groups to the left}.
  Thus, $\cK \, x \, y$ means $((\cK \, x) y)$ and we only write brackets when the grouping is \textit{not} to the left:
  $f \, (g \, x)$ and $f \, g \, x$ are different terms.

  On their own, terms just sit there. We allow them exactly one type of action, \textit{rewriting}: we write $M \leadsto
  N$ to say $M$ \textit{steps to} $N$, or one rewrite turns $M$ into $N$.
\end{definition}

\begin{definition}{1.2}[reduction]
  The relation $M \leadsto N$, is given by one of the rules:
  \begin{enumerate}
    \item Rule $\cK$ has nothing above the bar, so it holds outright. $\cK$ keeps the first of two arguments and drops
    the second.
    \begin{prooftree}
      \AxiomC{}
      \RightLabel{\scriptsize $\cK$}
      \UnaryInfC{$((\cK \, P) \, Q) \;\leadsto\; P$}
    \end{prooftree}

    \item Rule $\cS$ also has nothing above the bar, so it holds outright. $\cS$ hands its third argument to both of the
    others and applies the first result to the second.
    \begin{prooftree}
      \AxiomC{}
      \RightLabel{\scriptsize $\cS$}
      \UnaryInfC{$(((\cS \, P) \, Q) \, R) \;\leadsto\; ((P \, R) \, (Q \, R))$}
    \end{prooftree}

    \item Rule (app\textsubscript{1}) says that if $P \leadsto P'$ then $P \, Q \leadsto P' \, Q$: a step in the
    function is a step of the application.
    \begin{prooftree}
    \AxiomC{$P \leadsto P'$}
      \RightLabel{\scriptsize $\text{app}_1$}
      \UnaryInfC{$(P \, Q) \;\leadsto\; (P' \, Q)$}
    \end{prooftree}

    \item Rule (app\textsubscript{2}) says the same of the argument.
    \begin{prooftree}
      \AxiomC{$Q \leadsto Q'$}
      \RightLabel{\scriptsize $\text{app}_2$}
      \UnaryInfC{$(P \, Q) \;\leadsto\; (P \, Q')$}
    \end{prooftree}
  \end{enumerate}
\end{definition}

\begin{example}{1.4}[the identity] Here is the run of \textsf{I} on an argument:
  \[
  \begin{array}{r@{\;}l@{\qquad}l}
    \cS \, \cK \, \cK \, x & \leadsto \cK \, x \, (\cK \, x) & \text{the $\cS$
  step, with $P = Q = \cK$ and $R = x$} \\
                           & \leadsto x                      & \text{the $\cK$
  step: keep $x$, drop $\cK \, x$.}
  \end{array}
  \]
  The function that returns its argument unchanged isn't one of the two constants, but the language has it all the same.
  Whatever $x$ is, $\mathsf{I} \, x \leadsto^* x$.
\end{example}

Terms built from constants by application, together with rules for rewriting them, make a \textit{combinatory algebra},
and $\cK$ and $\cS$ are its basic combinators.
